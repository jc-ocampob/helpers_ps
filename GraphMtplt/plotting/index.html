    Referencing subplot¶

    SubPlot referencing is a array of the matrix representation where 0 is the defult selection and represents the left top. On the other hand n-1 references the right most bottom plot

    In [21]:
    Copied! 
    g = (
    hp.GraphMtplt(df)
    .plot(
        ncols=3,
        nrows=2,
        sharex=True,
        sharey=False,
        width_ratios=[0.2,0.3,0.5],
        height_ratios=[0.3,0.7]
    )
    .axis(2)
    .graph_line(
        series=[{"ticker":"nav"}]
    )
)
    g = ( hp.GraphMtplt(df) .plot( ncols=3, nrows=2, sharex=True, sharey=False, width_ratios=[0.2,0.3,0.5], height_ratios=[0.3,0.7] ) .axis(2) .graph_line( series=[{"ticker":"nav"}] ) )

    Adjusting subplot¶

    In [22]:
    Copied! 
    g = (
    hp.GraphMtplt(df)
    .plot(
        ncols=3,
        nrows=2,
        sharex=True,
        sharey=False,
        width_ratios=[0.2,0.3,0.5],
        height_ratios=[0.3,0.7]
    )
    .axis(2)
    .graph_line(
        series=[{"ticker":"nav"}]
    )
    .subplot_adjust(
        top=0.8,
        bottom=0.2,
        left=0.15,
        right=0.95
    )
)
